class Solution:
    def asteroidCollision(self, asteroids: List[int]) -> List[int]:
        n = len(asteroids)
        stack = []
        for i in range(n):
            if asteroids[i] > 0:
                stack.append(asteroids[i])
            else:
                a = asteroids[i]
                alive = True 
                while stack and stack[-1] > 0:
                    if stack[-1] < abs(a):
                        stack.pop()
                        continue
                    elif stack[-1] == abs(a):
                        stack.pop()
                        alive = False
                        break
                    else:
                        alive = False
                        break
                        
                if alive:
                    stack.append(a)
    
        return stack
