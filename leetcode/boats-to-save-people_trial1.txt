class Solution:
    def numRescueBoats(self, people: List[int], limit: int) -> int:
        n = len(people)
        people.sort()
        l = 0 
        r = n - 1
        boats = 0
        while l <= r:
            if people[l] + people[r] <= limit:
                boats+=1
                l+=1
                r-=1
            else:
                boats+=1
                r-=1
        return boats
