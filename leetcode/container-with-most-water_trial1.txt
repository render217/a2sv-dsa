class Solution:
    def maxArea(self, height: List[int]) -> int:
          n = len(height)
          max_a = float('-inf')
          l = 0
          r = n - 1
          while l < r:
              h = min(height[l],height[r])
              width = r - l
              max_a = max(max_a, width * h)
              if height[l] > height[r]:
                 r-=1
              else:
                 l+=1
          return max_a
