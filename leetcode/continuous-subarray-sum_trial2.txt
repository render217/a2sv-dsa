class Solution:
    def checkSubarraySum(self, nums: List[int], k: int) -> bool:
        n = len(nums)

        pref = [0] * (n+1)
        for i in range(n):
            pref[i+1] = pref[i] + nums[i]

        seen = {}
        is_good = False 

        for i,p_sum in enumerate(pref):
            rem = p_sum % k
            if rem in seen:
                if i - seen[rem] >= 2:
                    is_good = True
                    break
            else:
                seen[rem] = i

        return is_good
