class Solution:
    def countGood(self, nums: List[int], k: int) -> int:
        n = len(nums)
        count = Counter()
        left = 0
        pair_count = 0
        ans = 0
        for right in range(n):
            pair_count += count[nums[right]]
            count[nums[right]] += 1

            while pair_count >= k:
                ans += n - right
                count[nums[left]] -= 1
                pair_count -= count[nums[left]]
                left+=1
        
        return ans
