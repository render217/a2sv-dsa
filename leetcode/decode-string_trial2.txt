class Solution:
    def decodeString(self, s: str) -> str:

        stack = []

        for i,v in enumerate(s):
            if v != ']':
                stack.append(v)
            else:
                res = deque()

                while stack[-1] != '[':
                    res.appendleft(stack.pop())
                    
                stack.pop()

                num = deque()
                while stack and stack[-1].isdigit():
                    num.appendleft(stack.pop())

                value = "".join(res) * int("".join(num))
                stack.append(value)

        return "".join(stack)
