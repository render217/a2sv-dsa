class Solution:
    def duplicateZeros(self, arr: List[int]) -> None:
        """
        Do not return anything, modify arr in-place instead.
        """
        n = len(arr)
        zeros = arr.count(0)
        p1 = n - 1
        p2 = n - 1 + zeros
        
        while p1 >= 0:
            if arr[p1] == 0:
                # second zero
                if p2 < n: arr[p2] = 0
                p2 -= 1
                # first zero
                if p2 < n: arr[p2] = 0
                p2 -= 1
            else:
                if p2 < n: arr[p2] = arr[p1]
                p2 -= 1
            p1 -= 1

#               
# [1,  0,  2,  3,  0,  4,  5,  0, x, x, x ]   
#                              p1 
#                                      p2
