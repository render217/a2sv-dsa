class Solution:
    def findPeakElement(self, nums: List[int]) -> int:
        n = len(nums)
        l , h = 0, n - 1
        while l < h:
            m = l + (h-l) // 2
            if nums[m] < nums[m+1]:
                 l = m + 1
            else:
                h = m
        return l
