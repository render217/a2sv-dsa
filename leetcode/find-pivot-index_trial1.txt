class Solution:
    def pivotIndex(self, nums: List[int]) -> int:
        n = len(nums)
        sumLeft = [0] * n
        for i in range(1,n):
            sumLeft[i] = sumLeft[i-1]  + nums[i-1]
        
        sumRight = [0] * n
        for i in range(n-2,-1,-1):
            sumRight[i] = sumRight[i+1] + nums[i+1]
        
        ans = -1
        for i in range(n):
            if sumLeft[i] == sumRight[i]:
                ans = i
                break
                
        return ans
