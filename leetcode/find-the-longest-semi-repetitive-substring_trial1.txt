from collections import defaultdict
class Solution:
    def longestSemiRepetitiveSubstring(self, s: str) -> int:
       n = len(s)
       l = 0
       max_ = 0
       pairs = defaultdict(int)
       count = 0
       for r in range(n):
           if r > 0 and s[r] == s[r-1]:
                pairs[r]+= 1
                count += 1
           while count > 1:
              l+=1 
              if l in pairs:
                 count -= 1
                 del pairs[l]
            
           max_ = max(max_,r - l + 1)
       return max_
