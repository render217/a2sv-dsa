class Solution:
    def findRadius(self, houses: List[int], heaters: List[int]) -> int:
        houses.sort()
        heaters.sort()
        
        ans = 0

        for house in houses:
            idx = bisect_left(heaters,house)

            rr_left = float('inf')
            rr_right = float('inf')
            
            if idx > 0:
                rr_left = house - heaters[idx - 1]
            
            if idx < len(heaters):
                rr_right = heaters[idx] - house

            ans = max(ans,min(rr_left,rr_right))
        
        return ans
