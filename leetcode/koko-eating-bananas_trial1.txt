class Solution:
    def minEatingSpeed(self, piles: List[int], h: int) -> int:
        
        def canfinish(piles,h,k):
            total = 0
            for pile in piles:
                total += math.ceil(pile/k)
            return total <= h


        l = 1
        r = max(piles)

        while l < r:
            mid = l + (r - l) // 2

            if canfinish(piles,h,mid):
                r = mid
            else:
                l = mid + 1
            
        return l
