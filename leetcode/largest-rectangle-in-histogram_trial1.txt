class Solution:
    def largestRectangleArea(self, heights: List[int]) -> int:
        stack = []
        max_area = 0
        n = len(heights)

        for i in range(n):
            start = i
            while stack and stack[-1][1] > heights[i]:
                idx,h = stack.pop()
                max_area = max(max_area, h * (i - idx))
                start = idx
            
            stack.append((start,heights[i]))
        
        while stack:
            idx,h = stack.pop()
            max_area = max(max_area, h * (n - idx))
        
        return max_area
