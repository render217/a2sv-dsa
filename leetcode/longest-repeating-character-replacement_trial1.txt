class Solution:
    def characterReplacement(self, s: str, k: int) -> int:
        n = len(s)
        window = defaultdict(int)
        res = 0
        left = 0
        for right in range(n):
            window[s[right]] += 1
            while (right - left + 1) - max(window.values()) > k:
                window[s[left]] -= 1
                if window[s[left]] == 0:
                    del window[s[left]]
                left+=1
            res = max(res,right - left + 1)
        return res
