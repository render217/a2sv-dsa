class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        n = len(s)
        max_len = float('-inf')
        seen = set()
        start = 0
    
        for end in range(n): 
            while s[end] in seen:
                seen.remove(s[start])
                start += 1
            seen.add(s[end])
            max_len = max(max_len,end - start + 1)

        return 0 if max_len == float('-inf') else max_len
