class Node:
    def __init__(self, data=None,next_=None):
        self.val = data if data else {}
        self.next = next_

class LRUCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = {}  
        self.head = None
        self.last = None 

    def get(self, key: int) -> int:
        if key not in self.cache:
            return -1
        else:
            prev = None
            curr = self.head
            
            while curr:
                if key in curr.val:
                    # remove

                    # last
                    if curr == self.last:
                        return self.cache[key]
                    
                    # head
                    if prev is None:
                        self.head = curr.next
                    else:
                        prev.next = curr.next
                    

                    # append end
                    curr.next = None
                    if self.head is None:
                        self.head = curr
                        self.last = curr
                    else:
                        self.last.next = curr
                        self.last = curr
                    
                    return self.cache[key]

                prev = curr
                curr = curr.next
            

    def put(self, key: int, value: int) -> None:
        if key in self.cache:
            self.cache[key] = value
            
            prev = None
            curr = self.head
            
            while curr:
                if key in curr.val:
                    curr.val[key] = value
                    # remove

                    # last        
                    if curr == self.last:
                        return
                    
                    # head
                    if prev is None:
                        self.head = curr.next
                    else:
                        prev.next = curr.next
                    
                    # append end
                    curr.next = None
                    if self.head is None:
                        self.head = curr
                        self.last = curr
                    else:
                        self.last.next = curr
                        self.last = curr
                    return

                prev = curr
                curr = curr.next
        else:
            node = Node({key: value})

            if len(self.cache.keys()) >= self.capacity and self.head:
                # remove oldest
                old_key = list(self.head.val.keys())[0]
                del self.cache[old_key]
                
                self.head = self.head.next
                if self.head is None:
                    self.last = None
            
            if self.head is None:
                self.head = node
                self.last = node
            else:
                self.last.next = node
                self.last = node

            self.cache[key] = value
