from collections import Counter
class Solution:
    def maxVowels(self, s: str, k: int) -> int:
        vowels = set("aeiou")
        n = len(s)
        left = 0
        curr_sum = sum([1 for char in s[:k] if char in vowels])
        max_ = curr_sum
        for right in range(k,n):
           if s[right] in vowels:
             curr_sum += 1
           if s[right - k] in vowels:
             curr_sum -= 1
           max_ = max(max_, curr_sum)
        return max_
