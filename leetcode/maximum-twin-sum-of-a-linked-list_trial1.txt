# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def pairSum(self, head: Optional[ListNode]) -> int:
        nodes = []
        curr = head
        while curr:
            nodes.append(curr.val)
            curr = curr.next
        l , r = 0, len(nodes) - 1
        max_sum = float('-inf')
        while l < r:
            twin_sum = nodes[l] + nodes[r]
            max_sum = max(max_sum,twin_sum)
            l+=1
            r-=1
        return max_sum
