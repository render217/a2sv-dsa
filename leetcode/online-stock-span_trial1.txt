class StockSpanner:

    def __init__(self):
        self.stack = []

    def next(self, price: int) -> int:
        p = (price,1)
        
        while self.stack and self.stack[-1][0] <= price:
            popped_price, popped_span = self.stack.pop()
            p = (price , p[1] + popped_span)
        
        self.stack.append(p)

        return p[1]
