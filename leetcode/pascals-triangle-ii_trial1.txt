class Solution:
    def getRow(self, rowIndex: int) -> List[int]:
        if rowIndex == 0:
            return [1]

        last_row = self.getRow(rowIndex - 1)
        row = [1] * (rowIndex + 1)

        for i in range(1,rowIndex):
            row[i] = last_row[i - 1] + last_row[i]

        return row
