class Solution:
    def generate(self, numRows: int) -> List[List[int]]:
        pascal = [[1]]
        for i in range(1,numRows):
            temp = [0] + pascal[-1] + [0]
            build = []
            for j in range(len(temp)-1):
                build.append(temp[j] + temp[j+1])
            pascal.append(build)
        return pascal
