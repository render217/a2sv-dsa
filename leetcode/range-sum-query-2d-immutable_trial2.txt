class NumMatrix:

    def __init__(self, matrix: List[List[int]]):
        n, m = len(matrix),len(matrix[0])
        self.prefix = [[0 for _ in range(m+1)] for _ in range(n+1)]

        for i in range(1,n+1):
            for j in range(1,m+1):
                self.prefix[i][j] = matrix[i-1][j-1]
                self.prefix[i][j] += self.prefix[i-1][j]
                self.prefix[i][j] += self.prefix[i][j-1]
                self.prefix[i][j] -= self.prefix[i-1][j-1]

    def sumRegion(self, r1: int, c1: int, r2: int, c2: int) -> int:
            total = self.prefix[r2+1][c2+1]
            total += self.prefix[r1][c1]
            total -= self.prefix[r1][c2+1]
            total -= self.prefix[r2+1][c1]
            return total
