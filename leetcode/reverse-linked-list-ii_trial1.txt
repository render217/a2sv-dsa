# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
def reverse(head:ListNode):
        curr = head
        prev = None
        while curr:
            nextNode = curr.next
            curr.next = prev
            prev = curr
            curr = nextNode
        return prev

class Solution:
    def reverseBetween(self, head: Optional[ListNode], left: int, right: int) -> Optional[ListNode]:
            dummy = ListNode(0,head)
            
            l_cur = dummy
            for _ in range(left-1):
                l_cur = l_cur.next
            
            r_cur = dummy
            for _ in range(right):
                r_cur = r_cur.next
            
            last = r_cur.next

            r_cur.next = None

            new_head = reverse(l_cur.next)

            l_cur.next = new_head
            while l_cur.next:
                l_cur = l_cur.next
            
            l_cur.next = last

            return dummy.next
