# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def rotateRight(self, head: Optional[ListNode], k: int) -> Optional[ListNode]:
        if not head:
            return None
        
        tail = head
        count = 1

        while tail.next:
            count += 1
            tail = tail.next

        k = k % count
        if k == 0:
            return head

        i = count - k - 1
        temp = head
        while i > 0:
            temp = temp.next
            i -= 1

        new_head = temp.next

        tail.next = head

        temp.next = None

        return new_head
